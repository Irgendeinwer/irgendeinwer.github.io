{
  custom = {
    system = {
      kernel = "cachyos";
      boot.silent.enable = false;
      android.enable     = true;
    };

    desktop = {
      hyprland.enable    = true;
      gaming = {
        enable           = true;
        arkServer.enable = false;
      };
    };

    services = {
      backup = {
        enable             = true;
        autoTriggerOnPlug  = true;
      };
      hotspot = {
        enable             = false;
        ssid               = "6+7";
      };
      jellyfin.enable       = false;
      zapret.enable         = true;
      isolatedGaming.enable = false;
    };
  };
}
